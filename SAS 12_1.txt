function escapeHtml(str) {
  return String(str == null ? '' : str)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}

function typeLabel(t) {
  return {
    pg: 'Pilihan Ganda',
    table: 'Pilihan Ganda Kompleks',
    match: 'Menjodohkan',
    isian: 'Isian Singkat'
  }[t] || 'Soal';
}

function formatTanggal(d) {
  try {
    return d.toLocaleDateString('id-ID', {
      weekday: 'long', day: 'numeric', month: 'long', year: 'numeric'
    });
  } catch (e) {
    return d.toDateString();
  }
}

function formatJam(d) {
  try {
    return d.toLocaleTimeString('id-ID', { hour: '2-digit', minute: '2-digit' });
  } catch (e) {
    return '';
  }
}

function typeset(el) {
  if (window.MathJax && typeof window.MathJax.typesetPromise === 'function') {
    window.MathJax.typesetPromise([el]).catch(() => {});
  } else if (fallbackMode) {
    renderFallback(el);
  }
}

function texToHtml(tex) {
  let s = tex;
  s = s.replace(/\^\{([^{}]*)\}/g, '<sup>$1</sup>');
  s = s.replace(/\^(\w)/g, '<sup>$1</sup>');
  s = s.replace(/_\{([^{}]*)\}/g, '<sub>$1</sub>');
  s = s.replace(/_(\w)/g, '<sub>$1</sub>');

  let guard = 0;
  while (/\\[dt]?frac\{/.test(s) && guard++ < 12) {
    s = s.replace(/\\[dt]?frac\{([^{}]*)\}\{([^{}]*)\}/g,
      '<span class="math-frac"><span class="math-num">$1</span><span class="math-den">$2</span></span>');
  }

  s = s.replace(/\\sqrt\[([^\]]*)\]\{([^{}]*)\}/g,
    '<span class="math-root"><sup>$1</sup>√<span class="math-rad">$2</span></span>');
  s = s.replace(/\\sqrt\{([^{}]*)\}/g,
    '<span class="math-root">√<span class="math-rad">$1</span></span>');

  const MAP = {
    '\\times': '×', '\\cdot': '·', '\\div': '÷', '\\pm': '±',
    '\\pi': 'π', '\\alpha': 'α', '\\beta': 'β', '\\theta': 'θ', '\\circ': '°',
    '\\infty': '∞', '\\lim': 'lim', '\\sin': 'sin', '\\cos': 'cos', '\\tan': 'tan',
    '\\to': '→', '\\rightarrow': '→', '\\begin{pmatrix}': '(', '\\end{pmatrix}': ')', '\\\\': ' '
  };
  Object.keys(MAP).forEach(k => { s = s.split(k).join(MAP[k]); });
  return s.replace(/[{}]/g, '');
}

function renderFallback(root) {
  const walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT, null);
  const targets = [];
  let node;
  while ((node = walker.nextNode())) {
    if (/\$|\\\(|\\\[/.test(node.nodeValue)) targets.push(node);
  }
  targets.forEach(n => {
    let html = n.nodeValue
      .replace(/\$\$([\s\S]*?)\$\$/g, (m, ex) => '<span class="math-display">' + texToHtml(ex) + '</span>')
      .replace(/\\\[([\s\S]*?)\\\]/g, (m, ex) => '<span class="math-display">' + texToHtml(ex) + '</span>')
      .replace(/\$([^\$\n]+?)\$/g, (m, ex) => texToHtml(ex))
      .replace(/\\\(([\s\S]*?)\\\)/g, (m, ex) => texToHtml(ex));
    const span = document.createElement('span');
    span.innerHTML = html;
    n.parentNode.replaceChild(span, n);
  });
}

function initAnswers() {
  state.answers = questions.map(q => {
    if (q.type === 'pg') return null;
    if (q.type === 'table') return q.statements.map(() => null);
    if (q.type === 'match') return q.premises.map(() => null);
    return '';
  });
}

function isAnswered(i) {
  const q = questions[i];
  const a = state.answers[i];
  if (q.type === 'pg') return a !== null && a !== undefined;
  if (q.type === 'table') return Array.isArray(a) && a.every(v => v !== null);
  if (q.type === 'match') return Array.isArray(a) && a.every(v => v !== null);
  if (q.type === 'isian') return String(a || '').trim() !== '';
  return false;
}

function renderNav() {
  const grid = document.getElementById('navGrid');
  grid.innerHTML = questions.map((q, i) => {
    let cls = 'nav-btn';
    if (isAnswered(i)) cls += ' answered';
    if (state.flagged.has(i)) cls += ' flagged';
    if (i === state.current) cls += ' current';
    return '<button class="' + cls + '" data-idx="' + i + '">' + (i + 1) + '</button>';
  }).join('');

  const answeredCount = questions.filter((_, i) => isAnswered(i)).length;
  document.getElementById('progressText').textContent =
    answeredCount + ' / ' + questions.length + ' terjawab';
}

function renderQuestion() {
  const i = state.current;
  const q = questions[i];
  const card = document.getElementById('qCard');

  let html = '';

  html += '<div class="q-meta">';
  html += '<span class="q-number">Soal ' + (i + 1) + '</span>';
  html += '<span class="q-type">' + typeLabel(q.type) + '</span>';
  html += '<span class="q-topic">' + q.topic + '</span>';
  html += '</div>';

  html += '<div class="stimulus">' + q.question + '</div>';

  if (q.type === 'pg') {
    html += '<div class="options">';
    q.options.forEach((opt, oi) => {
      const sel = state.answers[i] === oi ? ' selected' : '';
      html += '<button class="option' + sel + '" data-opt="' + oi + '">' +
        '<span class="opt-key">' + LETTERS[oi] + '</span>' +
        '<span class="opt-text">' + opt + '</span>' +
        '</button>';
    });
    html += '</div>';
  }

  else if (q.type === 'table') {
    html += '<div class="table-wrap"><table class="bs-table"><thead><tr><th>Pernyataan</th>';
    q.columns.forEach(c => { html += '<th>' + c + '</th>'; });
    html += '</tr></thead><tbody>';

    q.statements.forEach((st, ri) => {
      html += '<tr><td class="stmt">' + st.text + '</td>';
      q.columns.forEach((c, ci) => {
        const sel = state.answers[i][ri] === ci ? ' selected' : '';
        html += '<td class="cell"><button class="cell-btn' + sel +
          '" data-row="' + ri + '" data-col="' + ci +
          '" aria-label="' + c + '"></button></td>';
      });
      html += '</tr>';
    });

    html += '</tbody></table></div>';
  }

  else if (q.type === 'match') {
    html += '<div class="match-list">';
    q.premises.forEach((p, ri) => {
      const val = state.answers[i][ri];
      const hasVal = val !== null && val !== undefined ? ' has-value' : '';
      html += '<div class="match-row">';
      html += '<div class="match-premise"><span class="premise-num">' + (ri + 1) + '</span><span>' + p + '</span></div>';
      html += '<select class="match-select' + hasVal + '" data-row="' + ri + '">';
      html += '<option value="">— Pilih jawaban —</option>';
      q.options.forEach((o, oi) => {
        const sel = val === oi ? ' selected' : '';
        html += '<option value="' + oi + '"' + sel + '>' + o + '</option>';
      });
      html += '</select></div>';
    });
    html += '</div>';
  }

  else if (q.type === 'isian') {
    html += '<div class="isian-wrap">';
    html += '<label class="isian-label" for="isianInput">Jawaban:</label>';
    html += '<input type="text" inputmode="decimal" autocomplete="off" class="isian-input" id="isianInput" ' +
      'value="' + escapeHtml(state.answers[i]) + '" placeholder="Ketik jawaban akhir (angka)">';
    if (q.unit) html += '<span class="isian-unit">' + q.unit + '</span>';
    if (q.hint) html += '<div class="isian-hint">' + q.hint + '</div>';
    html += '</div>';
  }

  card.innerHTML = html;

  document.getElementById('btnPrev').disabled = (i === 0);
  document.getElementById('btnNext').textContent =
    (i === questions.length - 1) ? 'Selesai & Lihat Hasil →' : 'Berikutnya →';
  document.getElementById('btnFlag').classList.toggle('active', state.flagged.has(i));

  renderNav();
  typeset(card);
}

function selectOption(oi) {
  if (state.finished) return;
  state.answers[state.current] = oi;
  document.querySelectorAll('.option').forEach(b => {
    b.classList.toggle('selected', Number(b.dataset.opt) === oi);
  });
  renderNav();
}

function selectCell(row, col) {
  if (state.finished) return;
  state.answers[state.current][row] = col;
  document.querySelectorAll('.cell-btn').forEach(b => {
    if (Number(b.dataset.row) === row) {
      b.classList.toggle('selected', Number(b.dataset.col) === col);
    }
  });
  renderNav();
}

function changeMatch(row, value) {
  if (state.finished) return;
  const arr = state.answers[state.current];
  const num = value === '' ? null : Number(value);

  if (num !== null) {
    for (let k = 0; k < arr.length; k++) {
      if (k !== row && arr[k] === num) arr[k] = null;
    }
  }
  arr[row] = num;

  document.querySelectorAll('.match-select').forEach(s => {
    const r = Number(s.dataset.row);
    s.value = arr[r] === null ? '' : String(arr[r]);
    s.classList.toggle('has-value', arr[r] !== null);
  });

  renderNav();
}

document.addEventListener('click', function (e) {
  const opt = e.target.closest('.option');
  if (opt) { selectOption(Number(opt.dataset.opt)); return; }

  const cell = e.target.closest('.cell-btn');
  if (cell) { selectCell(Number(cell.dataset.row), Number(cell.dataset.col)); return; }

  const navBtn = e.target.closest('.nav-btn');
  if (navBtn) {
    state.current = Number(navBtn.dataset.idx);
    renderQuestion();
    window.scrollTo({ top: 0, behavior: 'smooth' });
    return;
  }
});

document.addEventListener('change', function (e) {
  const sel = e.target.closest('.match-select');
  if (sel) { changeMatch(Number(sel.dataset.row), sel.value); }
});

document.addEventListener('input', function (e) {
  if (e.target.id === 'isianInput') {
    state.answers[state.current] = e.target.value;
    renderNav();
  }
});

function updateTimer() {
  const el = document.getElementById('timer');
  const txt = document.getElementById('timerText');
  const t = Math.max(0, state.timeLeft);

  const h = Math.floor(t / 3600);
  const m = Math.floor((t % 3600) / 60);
  const s = t % 60;
  const pad = n => String(n).padStart(2, '0');

  txt.textContent = pad(h) + ':' + pad(m) + ':' + pad(s);
  el.classList.toggle('warning', t <= 300);
}

function startTimer() {
  clearInterval(timerId);
  timerId = setInterval(function () {
    if (state.finished) return;
    state.timeLeft--;
    updateTimer();
    if (state.timeLeft <= 0) {
      state.timeLeft = 0;
      updateTimer();
      finishExam(true);
    }
  }, 1000);
}

function parseCandidates(raw) {
  const t = String(raw == null ? '' : raw).trim().replace(/\s/g, '');
  if (t === '') return [];
  const out = [];
  const push = v => { if (!isNaN(v) && isFinite(v)) out.push(v); };
  push(parseFloat(t));
  push(parseFloat(t.replace(/\./g, '').replace(',', '.')));
  push(parseFloat(t.replace(/,/g, '')));
  return out;
}

function scoreQuestion(q, ans) {
  if (q.type === 'pg') {
    return ans === q.answer ? 1 : 0;
  }
  if (q.type === 'table') {
    if (!Array.isArray(ans)) return 0;
    let benar = 0;
    for (let i = 0; i < q.statements.length; i++) {
      if (ans[i] === q.statements[i].answer) benar++;
    }
    return benar / q.statements.length;
  }
  if (q.type === 'match') {
    if (!Array.isArray(ans)) return 0;
    let benar = 0;
    for (let i = 0; i < q.answer.length; i++) {
      if (ans[i] === q.answer[i]) benar++;
    }
    return benar / q.answer.length;
  }
  if (q.type === 'isian') {
    const cands = parseCandidates(ans);
    return cands.some(v => Math.abs(v - q.answer) < 1e-6) ? 1 : 0;
  }
  return 0;
}

/* ================================================================
   13. FORMAT JAWABAN UNTUK HASIL
   ================================================================ */
function formatUserAnswer(q, a) {
  if (q.type === 'pg') {
    if (a === null || a === undefined) return '—';
    return LETTERS[a] + '. ' + q.options[a];
  }
  if (q.type === 'table') {
    return q.statements.map((st, ri) => {
      const v = a[ri];
      return '<div class="mini">' + st.text + ' → <b>' +
        (v === null || v === undefined ? '—' : q.columns[v]) + '</b></div>';
    }).join('');
  }
  if (q.type === 'match') {
    return q.premises.map((p, ri) => {
      const v = a[ri];
      return '<div class="mini">' + (ri + 1) + '. ' + p + ' → <b>' +
        (v === null || v === undefined ? '—' : q.options[v]) + '</b></div>';
    }).join('');
  }
  if (q.type === 'isian') {
    return (a && String(a).trim() !== '') ? escapeHtml(a) : '—';
  }
  return '—';
}

function formatKey(q) {
  if (q.type === 'pg') {
    return LETTERS[q.answer] + '. ' + q.options[q.answer];
  }
  if (q.type === 'table') {
    return q.statements.map(st => {
      return '<div class="mini">' + st.text + ' → <b>' + q.columns[st.answer] + '</b></div>';
    }).join('');
  }
  if (q.type === 'match') {
    return q.premises.map((p, ri) => {
      return '<div class="mini">' + (ri + 1) + '. ' + p + ' → <b>' +
        q.options[q.answer[ri]] + '</b></div>';
    }).join('');
  }
  if (q.type === 'isian') {
    return String(q.answer) + (q.unit ? ' ' + q.unit : '');
  }
  return '—';
}

function confirmFinish() {
  const belum = questions.filter((_, i) => !isAnswered(i)).length;
  const pesan = 'Yakin ingin menyelesaikan ujian?\n\n' +
    'Soal terjawab: ' + (questions.length - belum) + ' / ' + questions.length +
    (belum > 0 ? '\nMasih ada ' + belum + ' soal yang belum lengkap dijawab.' : '') +
    '\n\nJawaban tidak dapat diubah setelah ujian diselesaikan.';

  if (window.confirm(pesan)) finishExam(false);
}

function finishExam(auto) {
  if (state.finished) return;
  state.finished = true;
  state.finishedAt = new Date();
  clearInterval(timerId);

  document.getElementById('timer').classList.remove('warning');
  document.getElementById('timerText').textContent = 'SELESAI';

  showResult(auto);
}

const inputName = document.getElementById('inputName');
const nameError = document.getElementById('nameError');

function clearNameError() {
  inputName.classList.remove('error');
  nameError.classList.remove('show');
}

inputName.addEventListener('input', clearNameError);

inputName.addEventListener('keydown', function (e) {
  if (e.key === 'Enter') {
    e.preventDefault();
    startExam();
  }
});

function startExam() {
  const nama = inputName.value.trim().replace(/\s+/g, ' ');

  if (!nama) {
    inputName.classList.add('error');
    nameError.classList.add('show');
    inputName.focus();
    return;
  }

  clearNameError();
  state.name = nama;
  state.startedAt = new Date();

  const initial = nama.charAt(0).toUpperCase();
  document.getElementById('userName').textContent = nama;
  document.getElementById('userAvatar').textContent = initial;
  document.getElementById('userChip').style.display = 'flex';

  document.getElementById('sideName').textContent = nama;
  document.getElementById('sideAvatar').textContent = initial;
  document.getElementById('sidePeserta').style.display = 'flex';

  document.getElementById('startOverlay').classList.add('hidden');
  state.started = true;
  state.timeLeft = TOTAL_SECONDS;
  updateTimer();
  startTimer();
  renderQuestion();
}

document.getElementById('btnStart').addEventListener('click', startExam);
